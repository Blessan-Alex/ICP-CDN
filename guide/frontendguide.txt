```jsx
// Enhanced upload function using our dCDN backend
const uploadToDcdn = async (file) => {
  const reader = new FileReader();
  reader.onload = async () => {
    const bytes = new Uint8Array(reader.result);
    
    // Generate CID for the file
    const cid = await generateCID(file);
    
    // Upload to dCDN with automatic pinning
    const result = await backend.upload_content(
      cid,
      file.type,
      Array.from(bytes)
    );
    
    // Background pinning happens automatically
    console.log('File uploaded and pinned:', result);
  };
  reader.readAsArrayBuffer(file);
};
```

```jsx
// Smart content delivery component
const SmartContentDelivery = ({ cid }) => {
  const [content, setContent] = useState(null);
  const [cacheStatus, setCacheStatus] = useState('loading');
  
  const fetchContent = async () => {
    try {
      // Try to get from dCDN cache first
      const result = await backend.get_content(cid);
      setContent(result.Ok);
      setCacheStatus('cache_hit');
    } catch (error) {
      // Falls back to IPFS fetch automatically
      setCacheStatus('cache_miss');
    }
  };
  
  return (
    <div>
      <div className="cache-status">
        {cacheStatus === 'cache_hit' && <span className="text-green-500">⚡ Cache Hit</span>}
        {cacheStatus === 'cache_miss' && <span className="text-orange-500">🌐 IPFS Fetch</span>}
      </div>
      {/* Content display */}
    </div>
  );
};
```

```jsx
const ImageResizer = ({ cid, originalSize }) => {
  const [resizedImage, setResizedImage] = useState(null);
  const [targetWidth, setTargetWidth] = useState(originalSize.width);
  
  const resizeImage = async () => {
    const result = await backend.get_content_with_resize(cid, targetWidth);
    const blob = new Blob([result.Ok], { type: 'image/png' });
    setResizedImage(URL.createObjectURL(blob));
  };
  
  return (
    <div className="image-resizer">
      <div className="size-presets">
        <button onClick={() => setTargetWidth(200)}>Small (200px)</button>
        <button onClick={() => setTargetWidth(400)}>Medium (400px)</button>
        <button onClick={() => setTargetWidth(800)}>Large (800px)</button>
      </div>
      <input 
        type="range" 
        min="50" 
        max={originalSize.width} 
        value={targetWidth}
        onChange={(e) => setTargetWidth(parseInt(e.target.value))}
      />
      <button onClick={resizeImage}>Resize</button>
      {resizedImage && <img src={resizedImage} alt="Resized" />}
    </div>
  );
};
```
