```rust
// ===== CANISTER-TO-CANISTER COMMUNICATION FUNCTIONS =====
// These functions allow other canisters to directly interact with the dCDN

/// Canister-to-canister upload function
/// Allows other canisters to upload content directly with automatic cycles payment
#[ic_cdk::update]
async fn canister_upload(
    caller: Principal,
    content: Vec<u8>,
    content_type: String,
    cycles_payment: u128
) -> Result<String, String> {
    // Validate input
    if content.is_empty() {
        return Err("Content cannot be empty".to_string());
    }
    
    if content_type.is_empty() {
        return Err("Content type cannot be empty".to_string());
    }
    
    // Accept the cycles payment from the calling canister
    let cycles_available = ic_cdk::api::call::msg_cycles_available128();
    let cycles_accepted = ic_cdk::api::call::msg_cycles_accept128(cycles_payment.min(cycles_available));
    
    ic_cdk::print(format!("Canister upload: Accepted {} cycles from caller {}", cycles_accepted, caller));
    
    // Generate a CID for the content
    let cid = generate_cid_for_content(&content, &content_type);
    
    // Get or create user account for the calling canister
    let user_account = ACCOUNTS.with(|accounts| {
        let mut accounts = accounts.borrow_mut();
        let user_account = accounts.entry(caller).or_insert_with(|| UserAccount {
            user_principal: caller,
            cycles_balance: 0,
            tier: UserTier::Free,
            cache_usage_bytes: 0,
            pinata_enabled: false,
        });
        
        // Add the accepted cycles to the canister's balance
        user_account.cycles_balance = user_account.cycles_balance.saturating_add(cycles_accepted);
        
        user_account.clone()
    });
    
    // Create a cache entry for the uploaded content
    let cache_entry = CacheEntry {
        cid: cid.clone(),
        content_type: content_type.clone(),
        size: content.len() as u64,
        last_accessed_ts: ic_cdk::api::time(),
        bytes: content.clone(),
    };
    
    // Store in cache
    if let Err(e) = put_cache_entry(cid.clone(), cache_entry) {
        return Err(format!("Failed to cache uploaded content: {}", e));
    }
    
    // For paid tiers, also upload to Pinata for persistence
    if user_account.pinata_enabled {
        match upload_to_pinata(&content, &cid, &content_type, true).await {
            Ok(ipfs_hash) => {
                Ok(format!("Content uploaded and pinned to IPFS. CID: {}, IPFS Hash: {}", cid, ipfs_hash))
            }
            Err(e) => {
                Ok(format!("Content uploaded to cache. CID: {} (Pinata upload failed: {})", cid, e))
            }
        }
    } else {
        Ok(format!("Content uploaded to cache. CID: {} (Pinata not enabled for this tier)", cid))
    }
}

/// Canister-to-canister content retrieval function
#[ic_cdk::query]
fn canister_get_content(caller: Principal, cid: String) -> Result<Vec<u8>, String> {
    if cid.is_empty() {
        return Err("CID cannot be empty".to_string());
    }
    
    // Check if content is in cache
    if let Some(cache_entry) = get_cache_entry(&cid) {
        return Ok(cache_entry.bytes);
    }
    
    // Content not found in cache
    Err(format!("Content not found: {}", cid))
}

/// Canister-to-canister content retrieval with IPFS fallback
#[ic_cdk::update]
async fn canister_get_content_with_fallback(caller: Principal, cid: String) -> Result<Vec<u8>, String> {
    if cid.is_empty() {
        return Err("CID cannot be empty".to_string());
    }
    
    // First try to get from cache
    if let Some(cache_entry) = get_cache_entry(&cid) {
        return Ok(cache_entry.bytes);
    }
    
    // Cache miss, try IPFS fallback
    match fetch_from_ipfs_internal(&cid).await {
        Ok(content) => {
            // Cache the content for future requests
            let cache_entry = CacheEntry {
                cid: cid.clone(),
                content_type: "application/octet-stream".to_string(),
                size: content.len() as u64,
                last_accessed_ts: ic_cdk::api::time(),
                bytes: content.clone(),
            };
            
            // Try to cache the content
            let _ = put_cache_entry(cid.clone(), cache_entry);
            
            Ok(content)
        }
        Err(e) => {
            Err(format!("Content not found in cache or IPFS: {}", e))
        }
    }
}

/// Canister-to-canister bulk upload function
#[ic_cdk::update]
async fn canister_bulk_upload(
    caller: Principal,
    files: Vec<(Vec<u8>, String)>, // (content, content_type) pairs
    cycles_payment: u128
) -> Result<Vec<String>, String> {
    if files.is_empty() {
        return Err("Files list cannot be empty".to_string());
    }
    
    // Accept the cycles payment
    let cycles_available = ic_cdk::api::call::msg_cycles_available128();
    let cycles_accepted = ic_cdk::api::call::msg_cycles_accept128(cycles_payment.min(cycles_available));
    
    let mut results = Vec::new();
    
    for (content, content_type) in files {
        if content.is_empty() {
            results.push(Err("Content cannot be empty".to_string()));
            continue;
        }
        
        // Generate CID for this content
        let cid = generate_cid_for_content(&content, &content_type);
        
        // Get or create user account for the calling canister
        let user_account = ACCOUNTS.with(|accounts| {
            let mut accounts = accounts.borrow_mut();
            let user_account = accounts.entry(caller).or_insert_with(|| UserAccount {
                user_principal: caller,
                cycles_balance: 0,
                tier: UserTier::Free,
                cache_usage_bytes: 0,
                pinata_enabled: false,
            });
            
            user_account.clone()
        });
        
        // Create a cache entry
        let cache_entry = CacheEntry {
            cid: cid.clone(),
            content_type: content_type.clone(),
            size: content.len() as u64,
            last_accessed_ts: ic_cdk::api::time(),
            bytes: content.clone(),
        };
        
        // Store in cache
        match put_cache_entry(cid.clone(), cache_entry) {
            Ok(_) => {
                // For paid tiers, also upload to Pinata
                if user_account.pinata_enabled {
                    match upload_to_pinata(&content, &cid, &content_type, true).await {
                        Ok(ipfs_hash) => {
                            results.push(Ok(format!("{}:{}", cid, ipfs_hash)));
                        }
                        Err(e) => {
                            results.push(Ok(format!("{}:{}", cid, e)));
                        }
                    }
                } else {
                    results.push(Ok(cid));
                }
            }
            Err(e) => {
                results.push(Err(format!("Failed to cache content: {}", e)));
            }
        }
    }
    
    // Convert results to strings
    let string_results: Vec<String> = results.into_iter().map(|r| {
        match r {
            Ok(s) => format!("SUCCESS:{}", s),
            Err(e) => format!("ERROR:{}", e),
        }
    }).collect();
    
    Ok(string_results)
}

/// Canister-to-canister cycles deposit function
#[ic_cdk::update]
fn canister_deposit_cycles(caller: Principal) -> UserAccount {
    let cycles_available = ic_cdk::api::call::msg_cycles_available128();
    let cycles_accepted = ic_cdk::api::call::msg_cycles_accept128(cycles_available);
    
    ACCOUNTS.with(|accounts| {
        let mut accounts = accounts.borrow_mut();
        let user_account = accounts.entry(caller).or_insert_with(|| UserAccount {
            user_principal: caller,
            cycles_balance: 0,
            tier: UserTier::Free,
            cache_usage_bytes: 0,
            pinata_enabled: false,
        });
        
        // Add the accepted cycles to the user's cycles_balance
        user_account.cycles_balance = user_account.cycles_balance.saturating_add(cycles_accepted);
        
        user_account.clone()
    })
}

/// Helper function to generate CID for content
fn generate_cid_for_content(content: &[u8], content_type: &str) -> String {
    use std::collections::hash_map::DefaultHasher;
    use std::hash::{Hash, Hasher};
    
    let mut hasher = DefaultHasher::new();
    content.hash(&mut hasher);
    content_type.hash(&mut hasher);
    
    // Generate a pseudo-CID (in production, this would be a proper IPFS CID)
    format!("Qm{:x}{}", hasher.finish(), ic_cdk::api::time())
}
```

### **Create a Client Library for Other Canisters**

Now create a new file `src/icp_cdn_client/src/canister_client.rs`:

```rust
use candid::{CandidType, Deserialize, Principal};
use ic_cdk::api::call::call_with_payment;
use ic_cdk::api::call::CallResult;

#[derive(CandidType, Deserialize)]
pub struct CdnCanisterClient {
    cdn_canister_id: Principal,
}

impl CdnCanisterClient {
    pub fn new(cdn_canister_id: Principal) -> Self {
        Self { cdn_canister_id }
    }
    
    /// Upload content to the CDN canister
    pub async fn upload_content(
        &self,
        content: Vec<u8>,
        content_type: String,
        cycles_payment: u128
    ) -> CallResult<Result<String, String>> {
        call_with_payment(
            self.cdn_canister_id,
            "canister_upload",
            (ic_cdk::api::caller(), content, content_type, cycles_payment),
            cycles_payment,
        ).await
    }
    
    /// Get content from the CDN canister
    pub async fn get_content(&self, cid: String) -> CallResult<Result<Vec<u8>, String>> {
        ic_cdk::api::call::call(
            self.cdn_canister_id,
            "canister_get_content",
            (ic_cdk::api::caller(), cid),
        ).await
    }
    
    /// Get content with IPFS fallback
    pub async fn get_content_with_fallback(&self, cid: String) -> CallResult<Result<Vec<u8>, String>> {
        ic_cdk::api::call::call(
            self.cdn_canister_id,
            "canister_get_content_with_fallback",
            (ic_cdk::api::caller(), cid),
        ).await
    }
    
    /// Bulk upload multiple files
    pub async fn bulk_upload(
        &self,
        files: Vec<(Vec<u8>, String)>,
        cycles_payment: u128
    ) -> CallResult<Result<Vec<String>, String>> {
        call_with_payment(
            self.cdn_canister_id,
            "canister_bulk_upload",
            (ic_cdk::api::caller(), files, cycles_payment),
            cycles_payment,
        ).await
    }
    
    /// Estimate upload cost
    pub async fn estimate_upload_cost(&self, file_size_bytes: u64) -> CallResult<u128> {
        ic_cdk::api::call::call(
            self.cdn_canister_id,
            "canister_estimate_upload_cost",
            (file_size_bytes,),
        ).await
    }
    
    /// Estimate storage cost
    pub async fn estimate_storage_cost(&self, file_size_bytes: u64, hours: u64) -> CallResult<u128> {
        ic_cdk::api::call::call(
            self.cdn_canister_id,
            "canister_estimate_storage_cost",
            (file_size_bytes, hours),
        ).await
    }
    
    /// Get account information
    pub async fn get_account_info(&self) -> CallResult<Result<UserAccount, String>> {
        ic_cdk::api::call::call(
            self.cdn_canister_id,
            "canister_get_account_info",
            (ic_cdk::api::caller(),),
        ).await
    }
    
    /// Deposit cycles
    pub async fn deposit_cycles(&self, cycles_amount: u128) -> CallResult<UserAccount> {
        call_with_payment(
            self.cdn_canister_id,
            "canister_deposit_cycles",
            (ic_cdk::api::caller(),),
            cycles_amount,
        ).await
    }
}

#[derive(CandidType, Deserialize, Clone)]
pub struct UserAccount {
    pub user_principal: Principal,
    pub cycles_balance: u128,
    pub tier: UserTier,
    pub cache_usage_bytes: u64,
    pub pinata_enabled: bool,
}

#[derive(CandidType, Deserialize, Clone, PartialEq, Debug)]
pub enum UserTier {
    Free,
    Starter,
    Pro,
    Business,
}
```

```rust
async fn upload_chat_image(image_bytes: Vec<u8>) -> Result<String, String> {

        "image/jpeg".to_string(),

    match result {
        Ok(Ok(cid)) => Ok(cid),
        Ok(Err(e)) => Err(format!("Upload failed: {}", e)),
        Err(e) => Err(format!("Call failed: {:?}", e)),
    }
```
